cmake_minimum_required(VERSION 3.20)
project(packet_parser LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_library(packet_parser src/ipv4.cpp)
target_include_directories(packet_parser PUBLIC include)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(packet_parser PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(ipv4_test tests/ipv4_test.cpp)
  target_link_libraries(ipv4_test PRIVATE packet_parser)
  add_test(NAME ipv4_test COMMAND ipv4_test)
endif()
